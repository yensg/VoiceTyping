cmake_minimum_required(VERSION 3.16)
project(local_dictation_worker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_LIST_DIR}/../vendor/whisper.cpp/cmake")
set(WHISPER_BUILD_EXAMPLES ON CACHE BOOL "Build upstream common library" FORCE)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../vendor/whisper.cpp" whisper)
add_executable(dictation-worker worker.cpp)
target_include_directories(dictation-worker PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../vendor/whisper.cpp/examples")
target_link_libraries(dictation-worker PRIVATE common whisper json_cpp)
